∀ {X : Type u_1} {S : Type u_2} [inst : Fintype X] [inst_1 : Fintype S] [Nonempty S]
  (C : PhysicsOfConsciousness.ContinuingProcess X S),
  C.protocol.Positive →
    0 < C.temperature →
      C.protocol.LocalDetailedBalance C.temperature C.heat →
        ∀ (N k : ℕ),
          k < N →
            C.Sustains N →
              (∀ j ∈ Finset.range N, 0 ≤ (C.protocol.step j).meanHeat (C.heat j)) →
                PhysicsOfConsciousness.Chain.ActiveBound (C.protocol.step k) C.temperature
                  (C.heat k)
                  (C.stored + ∑ z, (C.protocol.law 0).p z * C.energy z -
                    ∑ z, (C.protocol.law N).p z * C.energy z)
